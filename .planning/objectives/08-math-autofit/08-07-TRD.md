```dart
// fit_text.dart
double computeFitFontSize(
  String text, BoxConstraints constraints, {
  required TextStyle style,
  double maxFontSize = 96, double minFontSize = 8,
}) {
  bool fits(double size) {
    final tp = TextPainter(
      text: TextSpan(text: text, style: style.copyWith(fontSize: size)),
      textDirection: TextDirection.ltr, maxLines: null,
    )..layout(maxWidth: constraints.maxWidth);
    return tp.height <= constraints.maxHeight && !tp.didExceedMaxLines
        && tp.width <= constraints.maxWidth;
  }
  if (fits(maxFontSize)) return maxFontSize;          // SHRINK-ONLY: already fits at max
  double lo = minFontSize, hi = maxFontSize;
  for (var i = 0; i < 12; i++) {                       // ~12 iters → sub-pixel precision
    final mid = (lo + hi) / 2;
    if (fits(mid)) { lo = mid; } else { hi = mid; }
  }
  return lo;
}
// FitText: a StatelessWidget using LayoutBuilder to get the real box, then Text at the fitted size.
```
