export-binary-smoke.sh shape (MIT header; set -euo pipefail; mirror scripts/check-chrome-export.sh's structure + skip discipline):
```bash
# Build to a temp path, write a tiny hand-built fixture deck, run pdf+png,
# assert magics. Exit code 3 from the binary == "no Chrome discoverable" ==
# documented clean skip (exit 0), matching every live-Chrome test's t.Skip.
tmp="$(mktemp -d)"; trap 'rm -rf "$tmp"' EXIT
go build -o "$tmp/eden-press-export" ./cmd/eden-press-export
cat > "$tmp/smoke.md" <<'DECK'
---
marp: true
---
# Smoke slide one
Plain text.
---
# Smoke slide two
More text.
DECK

# --- PDF ---
set +e; "$tmp/eden-press-export" "$tmp/smoke.md" -o "$tmp/out.pdf" --format pdf; rc=$?; set -e
if [ "$rc" -eq 3 ]; then echo "SKIP: no Chrome discoverable (binary exit 3) — smoke is a documented no-op here"; exit 0; fi
[ "$rc" -eq 0 ] || { echo "FAIL: pdf export exit $rc" >&2; exit 1; }
head -c 5 "$tmp/out.pdf" | grep -q '%PDF-' || { echo "FAIL: out.pdf missing %PDF- magic" >&2; exit 1; }

# --- PNG ---
mkdir -p "$tmp/pngs"
"$tmp/eden-press-export" "$tmp/smoke.md" -o "$tmp/pngs" --format png
count="$(find "$tmp/pngs" -name '*.png' | wc -l | tr -d ' ')"
[ "$count" -ge 1 ] || { echo "FAIL: no PNG files produced" >&2; exit 1; }
# PNG magic \x89PNG on the first file
first="$(find "$tmp/pngs" -name '*.png' | sort | head -n1)"
head -c 4 "$first" | grep -q $'\x89PNG' || { echo "FAIL: $first missing PNG magic" >&2; exit 1; }
echo "PASS: export-binary-smoke — pdf (%PDF-) + $count png(s) via eden-press-export"
```
