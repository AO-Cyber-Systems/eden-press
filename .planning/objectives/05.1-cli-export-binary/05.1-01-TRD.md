runExport pipeline (build in main.go — mirrors cmd/eden-press/convert.go's runConvert shape, but drives the exporters):
```go
// cmd/eden-press-export/main.go
func runExport(cmd *cobra.Command, args []string) error {
    arg := "-"; if len(args) == 1 { arg = args[0] }
    md, src, err := resolveInput(arg, cmd.InOrStdin()); if err != nil { return fail(exitRender, err) }
    opts := buildPressOptions(cmd)                              // press.Options straight from flags (no koanf)
    out, err := press.Render(md, opts); if err != nil { return fail(exitRender, err) }

    bp := browserPath(cmd)                                     // --browser-path flag ("" → Discover reads CHROME_PATH)
    if _, _, derr := chrome.Discover(chrome.DiscoverOptions{BrowserPath: bp}); derr != nil {
        return chromeUnavailableError(derr)                    // exit 3 + remedy message
    }
    sess, err := chrome.New(convert.Options{BrowserPath: bp}); if err != nil { return fail(exitRender, err) }
    defer sess.Close()

    switch f := cmd.Flags().Lookup("format").Value.String(); f {
    case "", "pdf":
        b, err := pdf.ToPDF(sess, out, pdf.Options{}); if err != nil { return fail(exitRender, err) }
        return writePDF(cmd, arg, src, b)
    case "png":
        imgs, err := png.ToImages(sess, out, png.Options{InlineSVG: opts.InlineSVG}); if err != nil { return fail(exitRender, err) }
        return writePNGs(cmd, arg, src, imgs)
    default:
        return fail(exitUsage, fmt.Errorf("unknown --format %q (want pdf|png)", f))
    }
}
```

const ( exitRender = 1; exitUsage = 2; exitNoChrome = 3 )
type cliError struct{ code int; err error }
func (e *cliError) Error() string { return e.err.Error() }
