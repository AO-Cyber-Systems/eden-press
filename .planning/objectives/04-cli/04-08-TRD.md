preview via pkg/browser behind an injectable seam (research CLI-04 / Don't-Hand-Roll):
```go
// cmd/eden-press/preview.go
import "github.com/pkg/browser"

// openURL is a package var so tests assert the target without launching a browser.
var openURL = func(url string) error { return browser.OpenURL(url) }

func runPreview(cmd *cobra.Command, args []string) error {
    arg := "-"; if len(args) == 1 { arg = args[0] }
    if arg == "-" { return fmt.Errorf("preview: cannot preview stdin (\"-\"); pass a file path") }
    // Convert to a temp standalone HTML (reuse 04-03), write, open file://.
    md, _, err := resolveInput(arg); if err != nil { return err }
    opts, err := buildOptions(cmd); if err != nil { return err }
    out, err := press.Render(md, opts); if err != nil { return err }
    doc := assembleHTML(out, htmlDocOptions{AutoFitScript: cfg.Bool("auto-fit-script")})
    f, err := os.CreateTemp("", "eden-press-*.html"); if err != nil { return err }
    if _, err := f.WriteString(doc); err != nil { return err }
    f.Close()
    return openURL("file://" + f.Name())
}
```
