```go
// cmd/eden-press/serve.go
func runServe(cmd *cobra.Command, args []string) error {
    root := "."; if len(args) == 1 { root = args[0] }
    absRoot, err := filepath.Abs(root); if err != nil { return err }
    hub := reload.NewHub()                         // 04-06 SSE plumbing (reused)
    mux := http.NewServeMux()
    mux.Handle("/__reload", hub)                    // SSE endpoint
    mux.HandleFunc("/", func(w http.ResponseWriter, r *http.Request) {
        p, ok := safeJoin(absRoot, r.URL.Path)      // TRAVERSAL GUARD (Pitfall 7)
        if !ok { http.Error(w, "forbidden", http.StatusForbidden); return }
        if isMarkdown(p) {
            md, err := os.ReadFile(p); if err != nil { http.NotFound(w, r); return }
            out, err := press.Render(string(md), buildServeOptions(cmd)); if err != nil { http.Error(w, err.Error(), 500); return }
            doc := assembleHTML(out, htmlDocOptions{                         // 04-03 pipeline (reused)
                AutoFitScript: cfg.Bool("auto-fit-script"),
                InjectScripts: []string{reload.ClientJS("/__reload")},       // 04-06 client (reused)
            })
            w.Header().Set("Content-Type", "text/html; charset=utf-8")
            io.WriteString(w, doc); return
        }
        http.ServeFile(w, r, p) // static files (images, css, etc.)
    })
    host := cfg.String("host"); if host == "" { host = "127.0.0.1" }
    port := cfg.Int("port"); if port == 0 { port = 8321 } // NON-8080 product default
    addr := net.JoinHostPort(host, strconv.Itoa(port))
    fmt.Fprintf(cmd.OutOrStdout(), "eden-press serving %s on http://%s\n", absRoot, addr)
    return http.ListenAndServe(addr, mux)
}

// safeJoin contains the request path under root (Pitfall 7 — defense-in-depth).
func safeJoin(root, urlPath string) (string, bool) {
    clean := filepath.Clean("/" + urlPath)          // strips ../ that would escape "/"
    joined := filepath.Join(root, clean)
    abs, err := filepath.Abs(joined); if err != nil { return "", false }
    rel, err := filepath.Rel(root, abs)
    if err != nil || rel == ".." || strings.HasPrefix(rel, ".."+string(os.PathSeparator)) { return "", false }
    return abs, true
}
```
Reuse note: if the server serves a directory index or a markdown path, it converts on EVERY request (no cache in v1 — matches Marp CLI). A live edit + save (with a watcher, or manual refresh) reflects immediately; the reload endpoint pushes refreshes.
