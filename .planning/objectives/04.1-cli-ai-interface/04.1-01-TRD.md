JSON envelope view — surface out.Model/out.Meta as `any` so cmd/eden-press NEVER imports chase/model (which would trip check-cli-imports.sh):

```go
// cmd/eden-press/format.go
type jsonEnvelope struct {
    HTML     string   `json:"html"`
    CSS      string   `json:"css"`
    Model    any      `json:"model"`    // out.Model (*model.Document) — marshals via its OWN json tags: {meta, sections, outline}
    Comments []string `json:"comments"` // out.Comments (flattened speaker notes)
    Meta     any      `json:"meta"`     // out.Meta (model.Meta) — convenience alias, top-level
}
func writeJSON(cmd *cobra.Command, out press.Output) error {
    env := jsonEnvelope{HTML: out.HTML, CSS: out.CSS, Model: out.Model, Comments: out.Comments, Meta: out.Meta}
    b, err := json.MarshalIndent(env, "", "  ")
    if err != nil { return cliFail(cmd, exitRender, fmt.Errorf("format json: marshal: %w", err)) }
    return writeOutputBytes(cmd, append(b, '\n'))
}
```

Exit-code + machine-readable-error machinery:
```go
const ( exitRender = 1; exitUsage = 2 )

type jsonError struct{ Error jsonErrorBody `json:"error"` }
type jsonErrorBody struct{ Code int `json:"code"`; Message string `json:"message"` }

type cliError struct { code int; err error; printed bool }
func (e *cliError) Error() string { return e.err.Error() }
func (e *cliError) Unwrap() error { return e.err }

// cliFail is the SINGLE failure sink for runConvert/emitFormat. When --format json
// is active it emits the JSON error envelope to stderr and marks printed=true so
// main does not re-print; otherwise it leaves plain-text printing to main.
func cliFail(cmd *cobra.Command, code int, err error) error {
    if cfg.String("format") == "json" {
        b, _ := json.MarshalIndent(jsonError{jsonErrorBody{Code: code, Message: err.Error()}}, "", "  ")
        fmt.Fprintln(cmd.ErrOrStderr(), string(b))
        return &cliError{code: code, err: err, printed: true}
    }
    return &cliError{code: code, err: err}
}
```

emitFormat — the ONE dispatch seam (04.1-02 fills the pptx case):
```go
func emitFormat(cmd *cobra.Command, out press.Output) error {
    switch f := cfg.String("format"); f {
    case "", "html":
        doc := assembleHTML(out, htmlDocOptions{AutoFitScript: cfg.Bool("auto-fit-script")})
        return writeOutput(cmd, doc) // existing string writer (convert.go)
    case "json":
        return writeJSON(cmd, out)
    case "pptx":
        // 04.1-02 wires pptx.ToPPTX here. Until then, a usage-style not-wired error.
        return cliFail(cmd, exitUsage, fmt.Errorf("format pptx: not wired yet (04.1-02)"))
    default:
        return cliFail(cmd, exitUsage, fmt.Errorf("unknown --format %q (want html|json|pptx)", f))
    }
}

// writeOutputBytes mirrors convert.go's writeOutput but for []byte (json/pptx):
// --output/-o path if set (os.WriteFile), else cmd.OutOrStdout().
func writeOutputBytes(cmd *cobra.Command, b []byte) error {
    if f := cmd.Flags().Lookup("output"); f != nil && f.Value.String() != "" {
        if err := os.WriteFile(f.Value.String(), b, 0o644); err != nil {
            return fmt.Errorf("writeOutputBytes: %w", err)
        }
        return nil
    }
    _, err := fmt.Fprint(cmd.OutOrStdout(), string(b))
    return err
}
```

```go
func runConvert(cmd *cobra.Command, args []string) error {
    arg := "-"; if len(args) == 1 { arg = args[0] }
    md, _, err := resolveInputFrom(arg, cmd.InOrStdin())
    if err != nil { return cliFail(cmd, exitRender, err) }
    opts, err := buildOptions(cmd)
    if err != nil { return cliFail(cmd, exitRender, err) }
    out, err := press.Render(md, opts)
    if err != nil { return cliFail(cmd, exitRender, err) }
    return emitFormat(cmd, out)
}
```
